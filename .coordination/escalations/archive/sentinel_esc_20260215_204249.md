```python
# error_trend_monitor.py
from collections import deque
import logging
from datetime import datetime

class ErrorTrendMonitor:
    def __init__(self, window=15):
        self.error_history = deque(maxlen=window)
        self.alert_threshold = 8
        self.growth_factor = 1.05  # 5%增长阈值
    
    def check_errors(self, current_errors):
        if len(self.error_history) >= self.alert_threshold:
            # 计算近期错误率
            recent = list(self.error_history)[-5:]
            current_rate = sum(recent) / len(recent)
            
            # 计算增长因子
            if len(recent) > 1:
                prev_rate = sum(list(self.error_history)[-6:]) / len(list(self.error_history)[-6:])
                growth = current_rate / prev_rate if prev_rate > 0 else 1
            else:
                growth = 1
            
            # 触发告警
            if current_rate > 2 or growth > self.growth_factor:
                logging.warning(f"⚠️ 错误率异常: {current_rate:.2f} errors/min (↑{growth:.2f}x增长)")
                send_alert("⚠️ 错误率显著增加", severity="警告")
        
        self.error_history.append(current_errors)
```
