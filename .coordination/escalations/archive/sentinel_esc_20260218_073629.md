```sql
SELECT DISTINCT run_id, location, event_data, created_at, updated_at
FROM trails
ORDER BY created_at;

-- 步骤 2: 替换原始表
DROP TABLE trails;
SELECT * INTO trails FROM trails_clean;

-- 步骤 3: 添加唯一约束
PRAGMA add_column(trails, location TEXT);
PRAGMA add_column(trails, run_id TEXT);
ALTER TABLE trails ADD UNIQUE(run_id, location);

-- 步骤 4: 创建幂等键
CREATE TABLE trail_entries (
    id INTEGER PRIMARY KEY,
    run_id TEXT NOT NULL,
    location TEXT NOT NULL,
    event_data JSONB,
    created_at DATETIME,
    updated_at DATETIME,
    UNIQUE(run_id, location)
);

-- 迁移现有数据（保留首次出现）
INSERT INTO trail_entries (run_id, location, event_data, created_at, updated_at)
SELECT run_id, location, JSON_OBJECT AS event_data, created_at, updated_at
FROM trails
DISTINCT ON (run_id, location) *;

-- 丢弃旧表
DROP TABLE trails;

-- 创建监控视图
CREATE VIEW trail_metrics AS
SELECT 
    COUNT(*) as total_trails,
    COUNT(DISTINCT run_id) as unique_runs,
    COUNT(DISTINCT location) as unique_locations,
    AVG(julianday(updated_at) - julianday(created_at)) as avg_age_days
FROM trail_entries;
```
